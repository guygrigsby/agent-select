// Probe A: MaxAgeSeconds near 2^53 must not overflow the millisecond bound.
func TestProbeFreshnessExtremeBound(t *testing.T) {
	sd := fixtureSnapshotData()
	rd := fixtureRequestData()
	rd.MaxAgeSeconds = canonicalIntMax // 9007199254740991 s
	dec, err := routeFixture(t, sd, rd)
	if err != nil {
		t.Fatalf("extreme bound must admit fresh offer: %v", err)
	}
	if dec.Selected == nil || dec.Selected.OfferID != "prov/alpha" {
		t.Fatalf("want alpha: %+v", dec.Selected)
	}
}

// Probe A2: age exactly equal to the bound is inside "age 0..bound".
func TestProbeFreshnessExactBoundary(t *testing.T) {
	sd := fixtureSnapshotData()
	rd := fixtureRequestData()
	// fixture alpha ConfirmedAt 2026-09-29 08:30 UTC (per TestRouteFreshness).
	rd.EffectiveAt = time.Date(2026, 9, 29, 8, 30, 0, 0, time.UTC)
	rd.MaxAgeSeconds = 0
	// 1-second bound, age 0.
	rd.MaxAgeSeconds = 1
	if _, err := routeFixture(t, sd, rd); err != nil {
		t.Fatalf("age 0 within bound must admit: %v", err)
	}
	// 1ms older than a 0-second bound is impossible; use age 1s > bound 0? bound 0 disables. Use bound 1, age 1500ms -> exceeded.
	rd.EffectiveAt = time.Date(2026, 9, 29, 8, 30, 1, 500000000, time.UTC)
	rd.EffectiveAt = rd.EffectiveAt.Truncate(time.Millisecond)
	dec, err := routeFixture(t, sd, rd)
	if err == nil {
		t.Fatal("age 1.5s must exceed a 1s bound")
	}
	if !containsStr(dec.Rejections[0].Reasons, "freshness_exceeded") {
		t.Fatalf("want freshness_exceeded: %v", dec.Rejections[0].Reasons)
	}
}

// Probe B: non-canonical ConfirmedAt (nanoseconds) must be rejected at
// construction with invalid_snapshot, not silently truncated or accepted.
func TestProbeConfirmedAtNonCanonicalRejected(t *testing.T) {
	sd := fixtureSnapshotData()
	sd.Offers[0].ConfirmedAt = time.Date(2026, 9, 29, 8, 30, 0, 123456789, time.UTC)
	_, err := NewSnapshot(sd)
	if err == nil {
		t.Fatal("nanosecond ConfirmedAt must be rejected")
	}
	if e, ok := err.(*Error); !ok || e.Code != CodeInvalidSnapshot {
		t.Fatalf("want invalid_snapshot, got %#v", err)
	}
	sd.Offers[0].ConfirmedAt = time.Date(2026, 9, 29, 8, 30, 0, 0, time.FixedZone("EST", -5*3600))
	if _, err := NewSnapshot(sd); err == nil {
		t.Fatal("non-UTC ConfirmedAt must be rejected")
	}
	if e, ok := err.(*Error); !ok || e.Code != CodeInvalidSnapshot {
		t.Fatalf("want invalid_snapshot, got %#v", err)
	}
}

// Probe C: enormous rates must give unknown cost with a warning, never a
// truncated cheap quote; strict mode rejects.
func TestProbeCostOverflowUnknown(t *testing.T) {
	sd := fixtureSnapshotData()
	rd := fixtureRequestData()
	// 63 chars: 59 nines + e+18 -> rate ~1e77 USD/Mtok; qty ~2000 overflows micros.
	huge := "99999999999999999999999999999999999999999999999999999999e18"
	if len(huge) > 64 {
		t.Fatal("probe rate too long")
	}
	for i := range sd.Offers {
		sd.Offers[i].Prices = []PriceData{
			{Component: "input_tokens", USDPerMillion: huge},
			{Component: "output_tokens", USDPerMillion: huge},
		}
	}
	rd.InputTokens, rd.OutputTokens = 2000, 1000
	dec, err := routeFixture(t, sd, rd)
	if err != nil {
		t.Fatalf("route failed: %v", err)
	}
	if dec.Selected == nil {
		t.Fatal("non-strict must admit unknown-cost candidate")
	}
	if dec.Selected.Cost.Known {
		t.Fatalf("overflow must be unknown cost: %+v", dec.Selected.Cost)
	}
	if !containsStr(dec.Selected.Warnings, "cost_unknown") {
		t.Fatalf("want cost_unknown warning: %v", dec.Selected.Warnings)
	}
	rd.StrictCost = true
	rd.BudgetKnown = true
	rd.BudgetMicros = 1000000
	dec2, err2 := routeFixture(t, sd, rd)
	if err2 == nil {
		t.Fatal("strict cost must reject unknown cost")
	}
	_ = dec2
}

// Probe D: cost mode must rank known-cost candidates ahead of unknown cost
// regardless of magnitude (unknown must not rank as free).
func TestProbeCostModeUnknownNotFree(t *testing.T) {
	sd := fixtureSnapshotData()
	rd := fixtureRequestData()
	rd.Mode = "cost"
	rd.BudgetKnown = false
	rd.BudgetMicros = 0
	rd.StrictCost = false
	rd.Runtime = []RuntimeOfferData{
		{ID: "a/cheap", Provider: "a", UpstreamID: "cheap", Eligible: true,
			Capabilities: []string{"text_input", "text_output"},
			Limits:  []LimitData{{Resource: "input_tokens", Maximum: 200000}, {Resource: "output_tokens", Maximum: 64000}},
			Prices:  []PriceData{{Component: "input_tokens", USDPerMillion: "1"}}}, // output_tokens missing
		{ID: "b/known", Provider: "b", UpstreamID: "known", Eligible: true,
			Capabilities: []string{"text_input", "text_output"},
			Limits:  []LimitData{{Resource: "input_tokens", Maximum: 200000}, {Resource: "output_tokens", Maximum: 64000}},
			Prices:  []PriceData{{Component: "input_tokens", USDPerMillion: "9000000"},
				{Component: "output_tokens", USDPerMillion: "9000000"}}},
	}
	rd.AllowUnscored = true
	dec, err := routeFixture(t, sd, rd)
	if err != nil {
		t.Fatal(err)
	}
	if dec.Selected.OfferID != "b/known" {
		t.Fatalf("unknown cost must not rank as free: %+v", dec.Selected)
	}
}

// Probe F: ceil once after summing, with fractional components.
func TestProbeCeilOnceAfterSum(t *testing.T) {
	total := new(big.Rat)
	for i := 0; i < 4; i++ {
		r := new(big.Rat)
		r.SetString("0.25")
		total.Add(total, r)
	}
	got := ceilRat(total)
	if got.Int64() != 1 {
		t.Fatalf("sum 1.0 -> 1, got %v", got)
	}
	total2 := new(big.Rat)
	r2, _ := new(big.Rat).SetString("0.2")
	total2.Add(total2, r2)
	total2.Add(total2, r2)
	if got := ceilRat(total2).Int64(); got != 1 {
		t.Fatalf("0.4 -> 1 (ceil once), got %d", got)
	}
}

// Probe G: decode round-trip of EncodeSnapshot output and hash stability.
func TestProbeEncodeDecodeHashStable(t *testing.T) {
	sd := fixtureSnapshotData()
	s, err := NewSnapshot(sd)
	if err != nil {
		t.Fatal(err)
	}
	enc, err := EncodeSnapshot(s)
	if err != nil {
		t.Fatal(err)
	}
	s2, err := DecodeSnapshot(enc)
	if err != nil {
		t.Fatal(err)
	}
	if s2.Hash() != s.Hash() {
		t.Fatal("hash must be stable across encode/decode")
	}
	enc2, err := EncodeSnapshot(s2)
	if err != nil {
		t.Fatal(err)
	}
	if string(enc) != string(enc2) {
		t.Fatal("encoding must round-trip byte-for-byte")
	}
	// Key-reordered but semantically identical JSON must be rejected (non-canonical).
	bad := []byte(`{"offers":[],"policy":"educated-v1","schema":1,"inputs":[],"distribution":"local","created_at":"2026-10-01T00:00:00.000Z","knowledge_cutoff":"2026-09-30T00:00:00.000Z"}`)
	if _, err := DecodeSnapshot(bad); err == nil {
		t.Fatal("non-canonical key order must be rejected")
	}
}
