package routing

import "testing"

func TestParentBaselineWireSuffixRejected(t *testing.T) {
	r, err := NewRequest(fixtureRequestData())
	if err != nil {
		t.Fatal(err)
	}
	b, err := CanonicalEncode(r.Data())
	if err != nil {
		t.Fatal(err)
	}
	for _, suffix := range []string{"}\n", "]\n", "} {\"credential\":\"secret\"}\n"} {
		d, err := DecodeRequestWire(append(append([]byte(nil), b...), suffix...))
		if err == nil {
			_, err = NewRequest(d)
		}
		if err == nil {
			t.Errorf("baseline accepts root suffix %q", suffix)
		}
	}
}
